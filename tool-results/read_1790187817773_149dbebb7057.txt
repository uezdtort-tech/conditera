   468→function simpleFaqMatch(message: string): { answer: string; quickReplies?: QuickReply[] } | null {
   469→  const lower = message.toLowerCase();
   470→  let best: { answer: string; quickReplies?: QuickReply[] } | null = null;
   471→  let bestScore = 0;
   472→  for (const topic of MINI_FAQ) {
   473→    let score = 0;
   474→    for (const kw of topic.keywords) {
   475→      if (lower.includes(kw)) score += kw.length > 5 ? 2 : 1;
   476→    }
   477→    if (score > bestScore) {
   478→      bestScore = score;
   479→      best = { answer: topic.answer, quickReplies: topic.quickReplies };
   480→    }
   481→  }
   482→  return bestScore > 0 ? best : null;
   483→}
