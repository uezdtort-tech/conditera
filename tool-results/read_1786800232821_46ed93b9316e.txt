    54→  const cakeBuilderOpen = useAppStore((s) => s.cakeBuilderOpen);
    55→  const setCakeBuilderOpen = useAppStore((s) => s.setCakeBuilderOpen);
    56→  const cakeBuilder = useAppStore((s) => s.cakeBuilder);
    57→  const updateCakeBuilder = useAppStore((s) => s.updateCakeBuilder);
    58→  const resetCakeBuilder = useAppStore((s) => s.resetCakeBuilder);
    59→  const navigate = useAppStore((s) => s.navigate);

    65→

    71→

    73→
    74→  const calculatePrice = () => {

    76→    const base = CAKE_BUILDER_OPTIONS.bases.find((b) => b.id === cakeBuilder.base);

    87→      price += (cakeBuilder.servings - 8) * 180;
    88→    }

    90→  };
    91→

    94→      case 0:
    95→        return !!cakeBuilder.eventType;
    96→      case 1:
    97→        return !!cakeBuilder.base;
    98→      case 2:
    99→        return !!cakeBuilder.filling;
   100→      case 3:
   101→        return !!cakeBuilder.coating;
   102→      case 4:
   103→        return true;
   104→      case 5:

   107→        return !!cakeBuilder.city && !!cakeBuilder.deliveryDate && !!cakeBuilder.deliveryType;
   108→      case 7:
   109→        return selectedConfectioners.length > 0;
   110→      default:
   111→        return true;
   112→    }

   119→    const deliveryCities = (c.location as { deliveryCities?: string[] })?.deliveryCities;

   122→  });

   125→    setSelectedConfectioners((prev) =>
   126→      prev.includes(id) ? prev.filter((c) => c !== id) : [...prev, id]

   128→  };

   136→      updateCakeBuilder({ step: step + 1 });

   140→        setCakeBuilderOpen(false);
   141→        setAuthModalOpen(true);
   142→        toast.info("Войдите, чтобы отправить запрос кондитерам");
   143→        return;
   144→      }

   150→          id: `neg_${Date.now()}_${Math.random().toString(36).slice(2, 8)}`,

   153→          customerName: user?.name || "Гость",
   154→          customerAvatar: user?.avatar,
   155→          confectionerId: confId,
   156→          confectionerName: conf?.businessName || "Кондитер",
   157→          confectionerAvatar: conf?.avatar,
   158→          originalRequest: {
   159→            eventType: cakeBuilder.eventType || "",
   160→            base: cakeBuilder.base || "",
   161→            filling: cakeBuilder.filling || "",
   162→            coating: cakeBuilder.coating || "",
   163→            decorations: cakeBuilder.decorations || [],
   164→            servings: cakeBuilder.servings || 8,
   165→            city: cakeBuilder.city || "",
   166→            deliveryDate: cakeBuilder.deliveryDate || "",
   167→            deliveryType: cakeBuilder.deliveryType || "delivery",
   168→            inscription: cakeBuilder.inscription || "",
   169→            comment: cakeBuilder.comment || "",
   170→            estimatedPrice: estimatedPrice + deliveryCost,
   171→          },

   173→          discountRequested: requestDiscount,
   174→          discountPercent: requestDiscount ? discountPercent : 0,
   175→          discountComment: requestDiscount ? discountComment : "",
   176→          quotedItems: [],
   177→          quotedTotal: 0,
   178→          quotedDeliveryCost: 0,
   179→          quotedPrepTime: "",
   180→          status: "pending_confectioner" as const,
   181→          revisions: [],
   182→          createdAt: new Date().toISOString(),
   183→          expiresAt: new Date(Date.now() + 48 * 60 * 60 * 1000).toISOString(),
   184→        };
   185→      });

   188→      useAppStore.setState({ negotiations: [...negotiations, ...newNegotiations] });
   189→

   192→        `Запрос отправлен ${selectedConfectioners.length} кондитер(ам)!`,
   193→        {

   225→    updateCakeBuilder({ dietary });

   229→  const deliveryCost = cakeBuilder.deliveryType === "delivery" ? calculateDelivery(estimatedPrice) : 0;

   232→    <Dialog open={cakeBuilderOpen} onOpenChange={(open) => !open && handleClose()}>

   277→              <div
   278→                key={i}
   279→                className={`h-1.5 flex-1 rounded-full transition-colors ${
   280→                  i <= step ? "bg-primary" : "bg-border"
   281→                }`}
   282→              />
   283→            ))}
   284→          </div>
   285→        </div>
   286→
   287→        {submitted ? (
   288→          /* Success screen */
   289→          <div className="relative z-10 p-8 text-center space-y-6 max-h-[70vh] overflow-y-auto">
   290→            <div className="h-20 w-20 rounded-full bg-emerald-100 flex items-center justify-center mx-auto">
   291→              <PartyPopper className="h-10 w-10 text-emerald-600" />
   292→            </div>
   293→            <div>
   294→              <h2 className="font-display text-2xl font-bold mb-2">
   295→                Запрос отправлен {selectedConfectioners.length} кондитер(ам)!
   296→              </h2>
   297→              <p className="text-muted-foreground max-w-md mx-auto">
   298→                Каждый кондитер получил ваш проект торта
   299→                {requestDiscount && discountPercent > 0 && (
   300→                  <> с запросом скидки {discountPercent}%</>
   301→                )}. Ожидайте предложений с ценой в течение 48 часов.
   302→                Согласование и оплата — в личном кабинете.
   303→              </p>
   304→            </div>
   305→
   306→            {/* Выбранные кондитеры */}
   307→            <div className="bg-card border border-border rounded-lg p-4 max-w-md mx-auto">
   308→              <div className="font-semibold mb-2 text-sm">Кому отправлено:</div>
   309→              <div className="space-y-2">
   310→                {selectedConfectioners.map((id) => {
   311→                  const conf = confectioners.find((c) => c.id === id);
   312→                  if (!conf) return null;
   313→                  return (
   314→                    <div key={id} className="flex items-center gap-2 text-sm">
   315→                      <img src={conf.avatar} alt="" className="w-6 h-6 rounded-full" loading="lazy" decoding="async" />
   316→                      <span>{conf.businessName}</span>
   317→                      <Badge variant="outline" className="text-[10px] ml-auto">
   318→                        ⏳ Ожидает ответа
   319→                      </Badge>
   320→                    </div>
   321→                  );
   322→                })}
   323→              </div>
   324→            </div>
   325→
   326→            <div className="bg-card border border-border rounded-lg p-4 text-left max-w-md mx-auto space-y-2">
   327→              <div className="font-semibold mb-2">Ваш запрос:</div>
   328→              <div className="text-sm space-y-1">
   329→                <div>
   330→                  <span className="text-muted-foreground">Мероприятие:</span>{" "}
   331→                  {CAKE_BUILDER_OPTIONS.eventTypes.find((e) => e.id === cakeBuilder.eventType)?.label}
   332→                </div>
   333→                <div>
   334→                  <span className="text-muted-foreground">Основа:</span>{" "}
   335→                  {CAKE_BUILDER_OPTIONS.bases.find((b) => b.id === cakeBuilder.base)?.label}
   336→                </div>
   337→                <div>
   338→                  <span className="text-muted-foreground">Начинка:</span>{" "}
   339→                  {CAKE_BUILDER_OPTIONS.fillings.find((f) => f.id === cakeBuilder.filling)?.label}
   340→                </div>
   341→                <div>
   342→                  <span className="text-muted-foreground">Покрытие:</span>{" "}
   343→                  {CAKE_BUILDER_OPTIONS.coatings.find((c) => c.id === cakeBuilder.coating)?.label}

   345→                {cakeBuilder.decorations.length > 0 && (
   346→                  <div>
   347→                    <span className="text-muted-foreground">Декор:</span>{" "}
   348→                    {cakeBuilder.decorations
   349→                      .map((d) => CAKE_BUILDER_OPTIONS.decorations.find((o) => o.id === d)?.label)
   350→                      .join(", ")}
   351→                  </div>
   352→                )}
   353→                <div>
   354→                  <span className="text-muted-foreground">Город:</span> {cakeBuilder.city}
   355→                </div>
   356→                <div>
   357→                  <span className="text-muted-foreground">Дата:</span> {cakeBuilder.deliveryDate}
   358→                </div>
   359→                <Separator className="my-2" />
   360→                <div className="flex justify-between font-semibold">
   361→                  <span>Оценка стоимости:</span>
   362→                  <span className="text-primary">
   363→                    ~{formatCurrency(estimatedPrice + deliveryCost)}
   364→                  </span>
   365→                </div>
   366→              </div>
   367→            </div>
   368→
   369→            <div className="flex gap-2 justify-center">
   370→              <Button onClick={handleClose} variant="outline">
   371→                Закрыть
   372→              </Button>
   373→              <Button
   374→                onClick={() => {
   375→                  handleClose();
   376→                  navigate("dashboard-customer", { tab: "negotiations" });
   377→                }}
   378→              >
   379→                Перейти к согласованию
   380→              </Button>
   381→            </div>
   382→          </div>
   383→        ) : (
   384→          <>
   385→            {/* Body — step content */}
   386→            <div className="relative z-10 flex-1 overflow-y-auto p-6 max-h-[60vh]">
   387→              {step === 0 && (
   388→                <StepContainer
   389→                  title="Какое мероприятие?"
   390→                  subtitle="Выберите тип события, чтобы мы подобрали подходящий торт"
   391→                >
   392→                  <div className="grid grid-cols-2 md:grid-cols-3 gap-3">
   393→                    {CAKE_BUILDER_OPTIONS.eventTypes.map((opt) => (
   394→                      <OptionCard
   395→                        key={opt.id}
   396→                        selected={cakeBuilder.eventType === opt.id}
   397→                        onClick={() => updateCakeBuilder({ eventType: opt.id as CakeBuilderState["eventType"] })}
   398→                      >
   399→                        <div className="text-3xl mb-2">{opt.icon}</div>
   400→                        <div className="font-medium text-sm">{opt.label}</div>
   401→                        <div className="text-xs text-muted-foreground mt-0.5">
   402→                          {opt.description}
   403→                        </div>
   404→                      </OptionCard>
   405→                    ))}
   406→                  </div>

   408→              )}
   409→
   410→              {step === 1 && (
   411→                <StepContainer
   412→                  title="Выберите основу торта"
   413→                  subtitle="Это определит текстуру и вкус вашего торта"
   414→                >
   415→                  <div className="grid grid-cols-1 md:grid-cols-2 gap-3">

   419→                        selected={cakeBuilder.base === opt.id}
   420→                        onClick={() => updateCakeBuilder({ base: opt.id as CakeBuilderState["base"] })}
   421→                      >
   422→                        <div className="flex items-start justify-between">
   423→                          <div>
   424→                            <div className="font-medium">{opt.label}</div>
   425→                            <div className="text-xs text-muted-foreground mt-0.5">
   426→                              {opt.description}
   427→                            </div>
   428→                          </div>
   429→                          {opt.priceBase > 0 && (
   430→                            <Badge variant="secondary" className="text-[10px]">
   431→                              +{formatCurrency(opt.priceBase)}
   432→                            </Badge>
   433→                          )}
   434→                        </div>
   435→                      </OptionCard>
   436→                    ))}
   437→                  </div>
   438→                </StepContainer>
   439→              )}
   440→
   441→              {step === 2 && (
   442→                <StepContainer
   443→                  title="Начинка"
   444→                  subtitle="Выберите любимую начинку для вашего торта"

   446→                  <div className="grid grid-cols-1 md:grid-cols-2 gap-3">
   447→                    {CAKE_BUILDER_OPTIONS.fillings.map((opt) => (

   449→                        key={opt.id}
   450→                        selected={cakeBuilder.filling === opt.id}
   451→                        onClick={() => updateCakeBuilder({ filling: opt.id })}

   458→                              <Badge variant="secondary" className="text-[10px] mt-0.5">
   459→                                +{formatCurrency(opt.price)}

   472→                  title="Покрытие"
